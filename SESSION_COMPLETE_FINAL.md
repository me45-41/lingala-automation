```bash
# Compilation
npm run build              # Compiler TypeScript
npm run dev                # Mode watch

# Exécution
npm start                  # CLI standard
npm run start:fast         # CLI avec cache
npm run start:fast --run   # Compilation + exécution

# Tests
npm run test               # Tests lexer
npm run test:all          # Tous les tests

# Cache
npm run cache:clean        # Vider le cache
npm run cache:stats        # Statistiques du cache

npm run build:exe          # Générer lingala.exe
npm run build:exe:all      # Générer pour toutes les plateformes
```
